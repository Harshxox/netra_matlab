## Features

### F1 — Grad-CAM (`generateGradCAM.m`)
MATLAB's `gradCAM` works directly on imported ONNX `dlnetwork` objects.
```matlab
function heatmap = generateGradCAM(net, img, grade)
    % net: imported ONNX dlnetwork
    % img: preprocessed image (H x W x 3 double)
    % grade: predicted class index (0-4)

    img224 = imresize(img, [224 224]);
    dlImg  = dlarray(single(img224), 'SSC');

    % Grad-CAM on the predicted class (grade+1 because MATLAB is 1-indexed)
    map = gradCAM(net, dlImg, grade + 1);

    % Resize heatmap back to original image size
    heatmap = imresize(map, [size(img,1) size(img,2)]);
    heatmap = mat2gray(heatmap);           % normalize to [0, 1]
end
```

### F2 — Heatmap Overlay (`overlayHeatmap.m`)
```matlab
function overlaidImg = overlayHeatmap(img, heatmap, alpha)
    if nargin < 3; alpha = 0.5; end

    % Apply jet colormap to heatmap
    jetMap = colormap(jet(256));
    heatIdx = round(heatmap * 255) + 1;
    heatIdx = max(1, min(256, heatIdx));
    colorHeatmap = reshape(jetMap(heatIdx(:), :), [size(heatmap) 3]);

    % Alpha blend with original image
    baseImg = im2double(img);
    overlaidImg = (1 - alpha) * baseImg + alpha * colorHeatmap;
    overlaidImg = im2uint8(overlaidImg);
end
```

### F3 — Evidence View (`buildEvidenceView.m`)
Combines Grad-CAM heatmap + lesion mask outlines on one image.
```matlab
function evidenceImg = buildEvidenceView(img, heatmap, masks)
    evidenceImg = overlayHeatmap(img, heatmap, 0.4);

    % Draw lesion contours on top
    colors = struct('MA',[255 0 0],'HE',[0 0 255],'EX',[255 255 0],'NV',[255 0 255]);
    fields = fieldnames(colors);
    for i = 1:numel(fields)
        f = fields{i};
        if isfield(masks, f)
            boundary = bwperim(masks.(f));
            for c = 1:3
                ch = evidenceImg(:,:,c);
                ch(boundary) = colors.(f)(c);
                evidenceImg(:,:,c) = ch;
            end
        end
    end
end
```

### F4 — Confidence Calibration (`calibrateConfidence.m`)
Temperature scalar T is produced by Colab and stored in `models/temperature.mat`.
```matlab
function calibratedProbs = calibrateConfidence(logits)
    persistent T
    if isempty(T)
        data = load('models/temperature.mat');
        T = data.temperature;
    end
    scaledLogits = logits / T;
    calibratedProbs = exp(scaledLogits) ./ sum(exp(scaledLogits));
end
```
If temperature.mat doesn't exist yet (Colab hasn't produced it), default T=1.0.

### F5 — PDF Report (`generateReport.m`)
Uses MATLAB Report Generator. Creates one-page annotated PDF per patient.
```matlab
function reportPath = generateReport(record, gradcamPath, overlayPath)
    import mlreportgen.report.*
    import mlreportgen.dom.*

    rpt = Report(fullfile('reports', [record.patientId '_report']), 'pdf');
    add(rpt, TitlePage('Title', 'Netra DR Screening Report', ...
        'Subtitle', ['Patient: ' record.patientId]));

    ch = Chapter('Title', 'Screening Result');

    % Header info
    add(ch, Paragraph(['Date: ' record.date ' | Eye: ' record.eye]));
    add(ch, Paragraph(['Quality: ' record.quality.status]));

    % Grade verdict (large, bold)
    verdict = Paragraph(sprintf('DR Grade: %d — %s', record.result.grade, record.result.gradeLabel));
    verdict.Bold = true;
    verdict.FontSize = '18pt';
    add(ch, verdict);

    refer = Paragraph(sprintf('Referable: %s | Confidence: %.1f%%', ...
        string(record.result.referable), record.result.confidence * 100));
    refer.Bold = true;
    refer.Color = ternary(record.result.referable, 'red', 'green');
    add(ch, refer);

    % Grad-CAM image
    add(ch, Paragraph('Grad-CAM Attention Map:'));
    add(ch, FormalImage(gradcamPath));

    % Lesion overlay
    add(ch, Paragraph('Lesion Overlay (red=MA, blue=HE, yellow=EX, magenta=NV):'));
    add(ch, FormalImage(overlayPath));

    % Lesion counts table
    tbl = Table({{'Lesion','Count'};
                 {'Microaneurysms', num2str(record.lesions.maCount)};
                 {'Hemorrhages',    num2str(record.lesions.heCount)};
                 {'Exudate area',   sprintf('%.1f%%', record.lesions.exudateAreaPct)};
                 {'Neovascularization', string(record.lesions.nvPresent)}});
    add(ch, tbl);

    % Routing
    add(ch, Paragraph(['Recommendation: ' record.routing]));

    % Disclaimer
    disc = Paragraph('This report is generated by a prototype screening tool and is NOT a certified medical device. All findings must be reviewed by a qualified ophthalmologist.');
    disc.Italic = true;
    disc.FontSize = '8pt';
    add(ch, disc);

    add(rpt, ch);
    close(rpt);
    reportPath = rpt.OutputPath;
end

function result = ternary(cond, a, b)
    if cond; result = a; else; result = b; end
end
```

### F6 — Explainability Pipeline Entry Point (`runExplainPipeline.m`)
```matlab
function [imagePaths, reportPath] = runExplainPipeline(net, img, record, masks)
    patientDir = fullfile('images', record.patientId);
    if ~exist(patientDir,'dir'); mkdir(patientDir); end

    % Grad-CAM
    heatmap    = generateGradCAM(net, img, record.result.grade);
    gradcamImg = overlayHeatmap(img, heatmap);
    gradcamPath = fullfile(patientDir, 'gradcam.png');
    imwrite(gradcamImg, gradcamPath);

    % Evidence view
    evidenceImg  = buildEvidenceView(img, heatmap, masks);
    evidencePath = fullfile(patientDir, 'evidence.png');
    imwrite(evidenceImg, evidencePath);

    imagePaths.gradcam  = gradcamPath;
    imagePaths.evidence = evidencePath;

    % PDF report
    reportPath = generateReport(record, gradcamPath, record.images.lesionOverlay);
end
```
